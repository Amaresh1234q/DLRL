import tensorflow as tf
import matplotlib.pyplot as plt
import numpy as np

dataset_path = "/kaggle/input/cat-and-dog"

train_ds = tf.keras.utils.image_dataset_from_directory(
    dataset_path,
    validation_split=0.2,
    subset="training",
    seed=123,
    image_size=(128, 128),
    batch_size=32
)

test_ds = tf.keras.utils.image_dataset_from_directory(
    dataset_path,
    validation_split=0.2,
    subset="validation",
    seed=123,
    image_size=(128, 128),
    batch_size=32
)

class_names = train_ds.class_names

print("Classes:", class_names)
print("Training images:", sum(len(images) for images, labels in train_ds))
print("Testing images :", sum(len(images) for images, labels in test_ds))
print("Total images :", sum(len(images) for images, labels in train_ds) + sum(len(images) for images, labels in test_ds))

model = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(128, 128, 3)),
    tf.keras.layers.Rescaling(1./255),
    tf.keras.layers.Conv2D(32, (3, 3), activation="relu"),
    tf.keras.layers.MaxPooling2D((2, 2)),
    tf.keras.layers.Conv2D(64, (3, 3), activation="relu"),
    tf.keras.layers.MaxPooling2D((2, 2)),
    tf.keras.layers.Flatten(),
    tf.keras.layers.Dense(64, activation="relu"),
    tf.keras.layers.Dense(1, activation="sigmoid")
])

model.compile(
    optimizer="adam",
    loss="binary_crossentropy",
    metrics=["accuracy"]
)

model.summary()

history = model.fit(
    train_ds,
    validation_data=test_ds,
    epochs=5
)

test_loss, test_accuracy = model.evaluate(test_ds)
print("Test Accuracy:", test_accuracy)

for images, labels in test_ds.take(1):
    first_image = images[0].numpy().astype("uint8")
    actual_label = labels[0].numpy()
    prediction = model.predict(tf.expand_dims(images[0], axis=0), verbose=0)[0][0]

plt.imshow(first_image)
plt.axis("off")
plt.show()

print("Actual:", "Dog" if actual_label == 1 else "Cat")
print("Predicted:", "Dog" if prediction >= 0.5 else "Cat")
